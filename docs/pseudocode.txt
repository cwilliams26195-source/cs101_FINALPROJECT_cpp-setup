START
    SET sports = ["Basketball", "Football", "Volleyball", "Softball", "Track and Field"]
    CREATE empty list
    SET opt = 0

    REPEAT
        DISPLAY menu:
            1. Register student
            2. View all students
            3. Search student by ID
            4. Registration summary
            5. Exit
        READ opt using getNum

        IF opt == 1 THEN
            READ name
            READ id
            READ age using getNum
            CALL printSports
            READ pick using getNum

            IF emptyInput(name) == 1 THEN
                DISPLAY "Name cannot be empty."
            ELSE IF emptyInput(id) == 1 THEN
                DISPLAY "ID cannot be empty."
            ELSE IF dupeId(id) == 1 THEN
                DISPLAY "That ID is already used."
            ELSE IF age < 5 OR age > 25 THEN
                DISPLAY "Age must be from 5 to 25."
            ELSE IF pick < 1 OR pick > 5 THEN
                DISPLAY "Pick a number from 1 to 5."
            ELSE
                SET chosen = sports[pick - 1]
                CREATE Student(name, id, age, chosen)
                ADD student to list
                DISPLAY "Added " + name + " to " + chosen + "."
            END IF

        ELSE IF opt == 2 THEN
            CALL showList
            IF list is empty THEN
                DISPLAY "No students yet."
            ELSE
                FOR each student in list DO
                    DISPLAY index, id, name, age, and sport
                END FOR
            END IF

        ELSE IF opt == 3 THEN
            READ id
            CALL findStudent(id)
            IF emptyInput(id) == 1 THEN
                DISPLAY "Enter an ID first."
            ELSE
                SET found = 0
                FOR each student in list DO
                    IF student.id == id THEN
                        DISPLAY id, name, age, and sport
                        SET found = 1
                        BREAK
                    END IF
                END FOR
                IF found == 0 THEN
                    DISPLAY "ID not found."
                END IF
            END IF

        ELSE IF opt == 4 THEN
            CALL summary
            SET totals = [0, 0, 0, 0, 0]
            FOR each student in list DO
                FOR j from 0 to 4 DO
                    IF student.sport == sports[j] THEN
                        totals[j] = totals[j] + 1
                    END IF
                END FOR
            END FOR
            DISPLAY "Total students: " + size of list
            FOR i from 0 to 4 DO
                DISPLAY sports[i] + ": " + totals[i]
            END FOR

        ELSE IF opt == 5 THEN
            DISPLAY "Bye!"

        ELSE
            DISPLAY "Invalid option."
        END IF

    UNTIL opt == 5
END
